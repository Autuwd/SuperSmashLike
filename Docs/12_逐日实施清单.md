```csharp
using UnityEngine;
using System.Collections.Generic;

namespace SuperSmashLike.UI
{
    public class UIManager : MonoBehaviour
    {
        public static UIManager Instance { get; private set; }

        [SerializeField] private HUDController hud;
        [SerializeField] private GameObject characterSelectPanel;
        [SerializeField] private GameObject pausePanel;
        [SerializeField] private GameObject resultPanel;

        private Stack<GameObject> panelStack = new();

        private void Awake()
        {
            if (Instance != null) { Destroy(gameObject); return; }
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }

        public void PushPanel(GameObject panel)
        {
            if (panelStack.Count > 0) panelStack.Peek().SetActive(false);
            panel.SetActive(true);
            panelStack.Push(panel);
        }

        public void PopPanel()
        {
            if (panelStack.Count > 0)
            {
                panelStack.Pop().SetActive(false);
                if (panelStack.Count > 0) panelStack.Peek().SetActive(true);
            }
        }

        public void ShowHUD() => PushPanel(hud.gameObject);
        public void HideHUD() { if (panelStack.Contains(hud.gameObject)) PopPanel(); }
    }
}
```

#### 5.2 创建 HUDController.cs

**文件位置**：`Assets/_Game/Scripts/UI/HUDController.cs`

```csharp
using UnityEngine;
using UnityEngine.UI;
using SuperSmashLike.Managers;

namespace SuperSmashLike.UI
{
    public class HUDController : MonoBehaviour
    {
        [System.Serializable]
        public struct PlayerHUD
        {
            public Text damageText;       // "85%"
            public Image stockIcon;       // 命数图标 (残机)
            public Image portrait;        // 角色头像
        }

        [SerializeField] private PlayerHUD[] playerHUDs;  // 4 个玩家槽位
        [SerializeField] private Text timerText;           // "3:00"
        [SerializeField] private Text matchStateText;       // "FIGHT!" / "KO!" / "GAME!"

        private MatchManager match;

        private void Start()
        {
            match = FindObjectOfType<MatchManager>();
        }

        private void Update()
        {
            // 方法清单：
            // UpdateTimerDisplay()    → 将 match.remainingTime 格式化为 "M:SS"
            // UpdatePlayerDisplay()   → 遍历 match.players, 更新伤害/命数
            // UpdateMatchState()      → 显示 "READY?" / "FIGHT!" / "{name} KO!" / "GAME!"
        }

        // === 需要实现的方法 ===

        // [实现目标] 格式化时间
        // 输入: float seconds → 输出: string "1:30"
        public void UpdateTimerDisplay()
        {
            if (match == null) return;
            int minutes = Mathf.FloorToInt(match.RemainingTime / 60);
            int secs = Mathf.FloorToInt(match.RemainingTime % 60);
            timerText.text = $"{minutes}:{secs:D2}";
        }

        // [实现目标] 更新玩家 HUD
        public void UpdatePlayerDisplay()
        {
            for (int i = 0; i < match.Players.Length && i < playerHUDs.Length; i++)
            {
                var fighter = match.Players[i];
                playerHUDs[i].damageText.text = $"{fighter.CurrentDamage:F0}%";
                playerHUDs[i].stockIcon.gameObject.SetActive(fighter.CurrentStocks > 0);
            }
        }

        // [实现目标] 显示KO提示
        public void ShowKOMessage(string fighterName)
        {
            matchStateText.text = $"{fighterName} KO!";
            Invoke(nameof(ClearMatchState), 1.5f);
        }

        private void ClearMatchState()
        {
            matchStateText.text = "";
        }
    }
}
```

            selectedIndices = new int[playerSlots.Length];
            SetupCharacterGrid();
            startButton.onClick.AddListener(StartMatch);
            startButton.interactable = false;
        }

        // [实现目标] 填充角色网格
        // 为每个 availableFighters[i] 创建一个格子
        private void SetupCharacterGrid()
        {
            for (int i = 0; i < availableFighters.Length; i++)
            {
                int index = i;  // 闭包捕获
                var slot = Instantiate(characterSlotPrefab, gridContainer);
                slot.GetComponentInChildren<Image>().sprite = availableFighters[i].portrait;
                slot.GetComponent<Button>().onClick.AddListener(() => SelectCharacter(0, index));
                // TODO: P2/P3/P4 的点击监听需要区分玩家
            }
        }

        // [实现目标] 玩家选择角色
        // playerIndex: 0~3, fighterIndex: 0~N
        public void SelectCharacter(int playerIndex, int fighterIndex)
        {
            if (playerSlots[playerIndex].isReady) return;
            selectedIndices[playerIndex] = fighterIndex;
            playerSlots[playerIndex].portrait.sprite = availableFighters[fighterIndex].portrait;
        }

        // [实现目标] 确认锁定（按 Enter/Start）
        public void ConfirmSelection(int playerIndex)
        {
            playerSlots[playerIndex].isReady = true;
            playerSlots[playerIndex].confirmIcon.SetActive(true);
            CheckAllReady();
        }

        private void CheckAllReady()
        {
            // 检查已连接的玩家是否都已锁定
            int connectedPlayers = GameManager.Instance.GetConnectedPlayerCount();
            bool allReady = true;
            for (int i = 0; i < connectedPlayers; i++)
            {
                if (!playerSlots[i].isReady) allReady = false;
            }
            startButton.interactable = allReady;
        }

        private void StartMatch()
        {
            // 存储选择数据供 BattleScene 使用
            for (int i = 0; i < selectedIndices.Length; i++)
            {
                PlayerPrefs.SetInt($"Player{i}_FighterIndex", selectedIndices[i]);
            }
            UnityEngine.SceneManagement.SceneManager.LoadScene(battleSceneName);
        }

public void OnGrabHit(FighterController target)
{
    if (stateMachine.CurrentState != FighterState.Grab)
    {
        grabbedTarget = target;
        stateMachine.TransitionToState(FighterState.Grab);
        target.stateMachine.TransitionToState(FighterState.Grabbed);
        target.grabbedBy = this;
        
        // 将目标拉到面前

public void PerformThrow(Vector2 direction)
{
    if (grabbedTarget == null) return;
    
    // 从 FighterData 获取投技数据
    AttackData throwAttack = fighterData.attacks[GetThrowIndex(direction)];
    
    // 应用伤害和击飞
    grabbedTarget.ApplyDamage(throwAttack.damage, throwAttack, this);
    
    // 释放目标
    grabbedTarget.grabbedBy = null;
    grabbedTarget = null;
    stateMachine.TransitionToState(FighterState.Idle);
    // 目标进入 Knockback
}

// [实现目标] 被抓方挣扎
public void StruggleEscape()
{

    struggleInputCount++;
    if (struggleInputCount >= struggleThreshold)

        // 挣脱
        grabbedBy.grabbedTarget = null;
        grabbedBy = null;
        stateMachine.TransitionToState(FighterState.Idle);
        // 推开对手
        Vector2 escapeDir = (transform.position - grabbedBy.transform.position).normalized;
        rb.AddForce(escapeDir * 8f, ForceMode2D.Impulse);

### 13.3 屏幕震动

```csharp
// [实现目标] 在 CameraManager 中添加：

private Vector3 originalPos;
private float shakeDuration = 0f;
private float shakeMagnitude = 0f;

// 调用方法：CameraManager.Instance.ShakeCamera(0.3f, 0.5f);
public void ShakeCamera(float duration, float magnitude)
{
    shakeDuration = duration;
    shakeMagnitude = magnitude;
    originalPos = transform.localPosition;
}

private void Update()
{
    if (shakeDuration > 0)
    {
        transform.localPosition = originalPos + Random.insideUnitSphere * shakeMagnitude;
        shakeDuration -= Time.deltaTime;
    }
    else
    {
        transform.localPosition = originalPos;
    }
}
```

**触发场景**：Smash 攻击命中时、击飞击杀时、破盾时。

### 13.4 Hitstop（打击暂停）

// [实现目标] 调整 Hitstop 参数
// 在 MatchManager 或全局配置中：

public void TriggerHitstop(float duration)
{
    StartCoroutine(HitstopCoroutine(duration));
}

private System.Collections.IEnumerator HitstopCoroutine(float duration)
{
    Time.timeScale = 0f;
    yield return new WaitForSecondsRealtime(duration);
    Time.timeScale = 1f;
}
